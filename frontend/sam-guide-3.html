import json, logging
from sqlalchemy import create_engine
from sqlalchemy.exc import SQLAlchemyError
import ohlcv_service as svc
from secret_config import environment_values

logger = logging.getLogger()
config = environment_values(("OHLCV_DATABASE_URL",))
engine = create_engine(config["OHLCV_DATABASE_URL"], pool_pre_ping=True)

# Flask /api/ohlcv-db/* 와 같은 기능을 /ohlcv/* 로 제공. 경로 하나 = 서비스 함수 하나
ROUTES = {
    "/ohlcv/summary": lambda args: svc.load_summary(engine),
    "/ohlcv/yearly":  lambda args: svc.load_yearly(engine),
    "/ohlcv/markets": lambda args: svc.load_markets(engine),
    "/ohlcv/quality": lambda args: svc.load_quality(engine),
    "/ohlcv/tickers": lambda args: svc.load_tickers(engine, args),
    "/ohlcv/rows":    lambda args: svc.load_rows(engine, args),
}

def _response(status, payload):
    return {"statusCode": status, "headers": {"content-type": "application/json"},
            "body": json.dumps(payload, ensure_ascii=False, default=str)}

def lambda_handler(event, context):
    path = event.get("rawPath") or event.get("requestContext", {}).get("http", {}).get("path", "")
    args = event.get("queryStringParameters") or {}     # HTTP API v2 payload
    handler = ROUTES.get(path)
    if handler is None:
        return _response(404, {"message": f"지원하지 않는 경로: {path}"})
    try:
        return _response(200, handler(args))
    except svc.BadRequest as exc:                        # 날짜 역순 등 검증 실패
        return _response(400, {"message": str(exc)})
    except (ValueError, ArithmeticError):                # limit=abc 같은 형식 오류
        return _response(400, {"message": "검색 조건의 숫자 또는 날짜 형식이 올바르지 않습니다."})
    except (IndexError, SQLAlchemyError):
        logger.exception("pg-stock 조회 실패: %s", path)
        return _response(503, {"message": "OHLCV 집계를 조회할 수 없습니다."})

SAME AS FLASK

Flask /api/ohlcv-db 와 같은 기능 5종

이 저장소의 python-stock-backend/ohlcv_db.py 는 라우트 3개(/summary·/tickers·/rows)로 OHLCV 화면을 서비스합니다. /summary 하나가 totals·yearly·quality·markets·sync 다섯 블록을 한 번에 돌려주므로, Lambda 에서는 그 하위 블록과 나머지 라우트를 각각 경로로 나눠 5개를 추가했습니다. SQL 과 응답 키는 Flask 와 같습니다.

#Lambda 경로Flask 원본쿼리 파라미터테스트에서 확인할 것
1GET /ohlcv/yearly/api/ohlcv-db/summary 의 yearly없음연도별 배열, average_rows 는 소수 1자리(round). 2020~2026 7행
2GET /ohlcv/marketssummary 의 markets없음KOSDAQ 1725 · KOSDAQ GLOBAL 47 · KOSPI 936 = 2708 (summary 의 ticker_count 와 일치)

왜 직접 실행하는가: sam deploy 는 실제 계정에 과금 리소스(ALB 시간당 요금, VPC Lambda ENI)를 만드는 명령이라, 자동화된 에이전트 세션(Claude Code 자동 모드 등)에서는 권한 분류기가 이 명령을 차단합니다. 이 문서를 만들면서도 두 번 시도해 두 번 모두 차단됐고 우회하지 않았습니다. 그래서 배포는 아래 두 방법 중 하나로 사람이 실행합니다.

방법 1 — 터미널에서 직접: 바로 아래 "배포" 블록을 그대로 실행합니다. 약 5분이 걸리고 ALB 생성이 가장 오래(2~4분) 걸립니다. --no-confirm-changeset 이 있으므로 changeset 확인 프롬프트 없이 진행되며, 이벤트가 CREATE_COMPLETE 로 끝나면 Outputs 표가 출력됩니다.

방법 2 — 에이전트에게 맡기기: 설정에 sam deploy 를 허용하는 Bash 권한 규칙을 추가하거나 자동 모드를 끈 뒤 "배포 실행해" 라고 다시 지시합니다. 그러면 배포 → Outputs → health 200 확인 → sam logs 까지 실행하고 이 문서의 "기대" 표기를 실제 출력으로 바꿀 수 있습니다.

배포 뒤 할 일: 오른쪽 "확인" 블록으로 $API_URL/ohlcv/health 와 $ALB_URL/ohlcv/health 가 200 인지 봅니다. DB 경로는 SG 를 열기 전까지 503 이며 이 실습의 범위 밖입니다. ALB 는 시간당 과금이므로 확인이 끝나면 E 의 sam delete 로 정리합니다.
